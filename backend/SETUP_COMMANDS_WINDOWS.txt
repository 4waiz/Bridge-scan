PRINTMIND - NEW MACHINE SETUP (WINDOWS POWERSHELL)
==================================================

Use this file as the copy/paste checklist on a new PC.

Prerequisites (install manually first)
--------------------------------------
1) Python 3.10.x (64-bit)
2) Git for Windows
3) PrusaSlicer
4) Bambu Studio


A) Create virtual environment and install dependencies
------------------------------------------------------
# If scripts are blocked in PowerShell:
Set-ExecutionPolicy -Scope Process -ExecutionPolicy Bypass

cd C:\printmind
py -3.10 -m venv venv310
.\venv310\Scripts\Activate.ps1

python -m pip install --upgrade pip setuptools wheel

# Install GPU PyTorch (recommended for fast generation)
pip install torch torchvision torchaudio --index-url https://download.pytorch.org/whl/cu126

# Install local Shap-E package from this repo
pip install -e .\shap-e

# Install backend deps
pip install -r .\printmind_app\backend\requirements.txt

# Optional sanity checks
python -c "import torch; print('cuda=', torch.cuda.is_available()); print(torch.cuda.get_device_name(0) if torch.cuda.is_available() else 'NO_GPU')"
python -c "import shap_e, fastapi, trimesh, bambulabs_api; print('python deps OK')"

E) Generate model -> estimate -> send print (Terminal 2)
--------------------------------------------------------
# 1) Generate STL from prompt
$genBody = @{
  prompt = "A smooth rounded vase with narrow neck"
  quality_preset = "balanced"
} | ConvertTo-Json

$gen = Invoke-RestMethod -Uri "http://localhost:8000/api/generate" -Method Post -ContentType "application/json" -Body $genBody
$gen | ConvertTo-Json -Depth 8

# 2) Estimate print first
$estBody = @{
  stl_path = $gen.stl_path
  printer_name = "BRIDGE2"
  mode = "decorative"
  auto_start = $false
  use_ams = $true
  ams_slot = 3
  bed_leveling = $true
  flow_cali = $true
  vibration_cali = $true
  layer_inspect = $false
  timelapse = $false
  bed_type = "Textured PEI Plate"
  nozzle_mm = 0.4
  nozzle_type = "hardened_steel"
  support_mode = "auto"
  support_type = "tree(auto)"
  support_on_build_plate_only = $false
} | ConvertTo-Json

$est = Invoke-RestMethod -Uri "http://localhost:8000/api/slice_estimate" -Method Post -ContentType "application/json" -Body $estBody
$est | ConvertTo-Json -Depth 8

# 3) Send + auto-start print
$printBody = @{
  stl_path = $gen.stl_path
  printer_name = "BRIDGE2"
  mode = "decorative"
  auto_start = $true
  use_ams = $true
  ams_slot = 3
  bed_leveling = $true
  flow_cali = $true
  vibration_cali = $true
  layer_inspect = $false
  timelapse = $false
  bed_type = "Textured PEI Plate"
  nozzle_mm = 0.4
  nozzle_type = "hardened_steel"
  support_mode = "auto"
  support_type = "tree(auto)"
  support_on_build_plate_only = $false
} | ConvertTo-Json

$res = Invoke-RestMethod -Uri "http://localhost:8000/api/slice_send" -Method Post -ContentType "application/json" -Body $printBody
$res | ConvertTo-Json -Depth 10

# 4) Monitor status
1..20 | ForEach-Object {
  $s = Invoke-RestMethod "http://localhost:8000/api/printer_status?printer_name=BRIDGE2"
  "{0} state={1} pct={2}% rem={3}m err={4} task={5}" -f (Get-Date -Format "HH:mm:ss"), $s.gcode_state, $s.percent, $s.remaining_minutes, $s.print_error_hex, $s.subtask_name
  Start-Sleep -Seconds 2
}


F) Useful recovery commands
---------------------------
# Stop current job
$ctl = @{ printer_name = "BRIDGE2" } | ConvertTo-Json
Invoke-RestMethod -Uri "http://localhost:8000/api/printer_stop" -Method Post -ContentType "application/json" -Body $ctl | ConvertTo-Json -Depth 10

# Resume current job
Invoke-RestMethod -Uri "http://localhost:8000/api/printer_resume" -Method Post -ContentType "application/json" -Body $ctl | ConvertTo-Json -Depth 10
